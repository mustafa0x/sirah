# أوائل المهاجرين يخرجون من مكة

chapter: hch_early_migration
stage: setting
prefix: sp_earlier_exp_

**Question:** كيف شقّ أصحاب النبي ﷺ طريقهم من مكة إلى المدينة قبله؟

## Overview

كان المؤمنون بمكة يُشتَمون ويُؤذَون، فأذن لهم النبي ﷺ بالخروج، وأخبرهم أنّ الله قد جعل لهم إخوانًا ودارًا يأمنون فيها.[[h_146_237_permission:p1–p2 | الواقدي، عن الزهري وعن عروة عن عائشة]][[h_9783_994_early_migrants:p1 | النبي ﷺ يأمر أصحابه بالهجرة]] غير أنّ الخروج لم يكن هيّنًا: فقد تتمزّق الأسرة قبل أن تجاوز البلد، وقد يردّ الرجلَ عن طريقه نذرُ أمّه، وقد يكون كلُّ ما جمعه المرء ثمنًا للطريق.[[h_23833_492_prep_cont:p1–p2 | أم سلمة، من طريق ابن إسحاق عن أبيه عن سلمة بن عبد الله]][[h_23833_497_prep_786:p1, h_23833_498_prep_cont:p1 | عمر]][[h_23833_500_prep_791:p2–p3 | ابن هشام: «ذُكر لي»، عن أبي عثمان النهدي: «بلغني»]] وتسلّلت البيوت بيتًا بعد بيت، وقريشٌ ترى الدورَ تخلو من أهلها.[[h_23833_493_prep_781:p1, h_23833_494_prep_cont:p1–p7 | ابن إسحاق: بنو جحش، وأبو أحمد، والدار الخالية، وعتبة وأبو جهل]]

## بيتٌ يتفرّق

وقبل البيعة الكبرى بالعقبة كان رجلٌ قد خرج. ذلك أبو سلمة بن عبد الأسد، ابن عمّة رسول الله ﷺ، وكان قد هاجر قبلُ إلى أرض الحبشة وخرج معه بامرأته.[[h_9862_213_ishaq_early_abusalama:p2 | رواية يونس بن بكير]] فلمّا رجع إلى مكة آذته قريش، فلمّا بلغه أنّ لهم بالمدينة إخوانًا عزم على اللحاق بهم، فكان أوّلَ من هاجر إليها من أصحاب رسول الله ﷺ من قريش.[[h_23833_491_prep_780:p1, h_930_737_prep_abu_salama_date:p1 | ابن إسحاق]]

وقد حدّثت امرأته أمّ سلمة بما جرى. رحل أبو سلمة بعيره وحملها عليه، وفي حجرها ابنهما الصغير سلمة، ثم خرج يقود بها البعير. فاعترضه رجالٌ من قومها بني المغيرة، فقالوا: «هذه نفسك غلبتنا عليها، أرأيتَ صاحبتك هذه؟ علامَ نتركك تسير بها في البلاد؟» ونزعوا خطام البعير من يده، وأخذوها منه. فغضب رهط أبي سلمة، بنو عبد الأسد، وطالبوا بالصبيّ، وتجاذبه الفريقان بينهم حتى خلعوا يده. فانطلق به قوم أبيه، وحبسها قومها عندهم، ومضى أبو سلمة إلى المدينة.[[h_23833_492_prep_cont:p1–p2 | أم سلمة، من طريق ابن إسحاق عن أبيه عن سلمة بن عبد الله]]

قالت: «ففرَّقَ بيني وبين زوجي وبين ابني». فكانت تخرج كلّ غداة فتجلس بالأبطح تبكي، حتى مضت سنةٌ أو قريبٌ منها.[[h_23833_492_prep_cont:p1–p2 | أم سلمة، من طريق ابن إسحاق عن أبيه عن سلمة بن عبد الله]]

## «وهي يثرب»

ثم كانت الليلة التي بايع فيها سبعون رجلًا من أهل المدينة رسولَ الله ﷺ بالعقبة، فطابت نفسه، وقد جعل الله له مَنَعةً وقومًا أهلَ حرب.[[h_146_237_permission:p1–p2 | الواقدي، عن الزهري وعن عروة عن عائشة]][[h_146_238_early_migrants:p1 | الواقدي]] لكنّ قريشًا علمت حينئذٍ أنّ المؤمنين عازمون على الخروج، فاشتدّت عليهم أكثر ممّا كانت. واستأذن الأصحاب في الهجرة، فخرج إليهم النبي ﷺ ذات يوم مسرورًا فقال: «قد أُخبرتُ بدار هجرتكم وهي يثرب، فمن أراد الخروج فليخرج إليها».[[h_146_237_permission:p1–p2 | الواقدي، عن الزهري وعن عروة عن عائشة]] وقال لهم: «إنّ الله عزّ وجلّ قد جعل لكم إخوانًا ودارًا تأمنون فيها».[[h_9783_994_early_migrants:p1 | النبي ﷺ يأمر أصحابه بالهجرة]] فجعلوا يتجهّزون ويتواسَون، ويخرجون مستخفين، جماعةً بعد جماعة.[[h_146_237_permission:p1–p2 | الواقدي، عن الزهري وعن عروة عن عائشة]][[h_9783_994_early_migrants:p1 | النبي ﷺ يأمر أصحابه بالهجرة]]

وخرجت بيوتٌ بأكملها. احتمل عبد الله بن جحش أهلَه وأخاه أبا أحمد، وكان أبو أحمد رجلًا ضرير البصر شاعرًا، يطوف مكة أعلاها وأسفلها بغير قائد؛ وخرج معهم قومهم جميعًا، رجالُهم ونساؤهم.[[h_23833_493_prep_781:p1, h_23833_494_prep_cont:p1–p7 | ابن إسحاق: بنو جحش، وأبو أحمد، والدار الخالية، وعتبة وأبو جهل]][[h_23833_495_prep_cont:p1 | ابن إسحاق]] ومرّ عتبة بن ربيعة والعباس بن عبد المطلب وأبو جهل ذات يوم بدار بني جحش، فإذا هي مغلقة خاوية، تخفق أبوابها يبابًا، ليس فيها ساكن. فتنفّس عتبة الصُّعَداء وقال: «أصبحت دار بني جحش خلاءً من أهلها!» فقال أبو جهل: «وما تبكي عليه من قُلِّ بنِ قُلّ؟» ثم قال: «هذا عملُ ابنِ أخي هذا، فرّق جماعتنا، وشتّت أمرنا، وقطع بيننا».[[h_23833_493_prep_781:p1, h_23833_494_prep_cont:p1–p7 | ابن إسحاق: بنو جحش، وأبو أحمد، والدار الخالية، وعتبة وأبو جهل]]

## حُبسوا عن الطريق

وتواعد عمر بن الخطاب وعيّاش بن أبي ربيعة وهشام بن العاص على الخروج، وجعلوا موعدهم التناضب، وقالوا: «أيُّنا لم يُصبح عندها فقد حُبس، فليمضِ صاحباه». فأصبح عمر وعيّاش عند التناضب، وحُبس عنهما هشام، وفُتن فافتُتن.[[h_23833_497_prep_785:p1–p3, h_7478_982_ra_umar_hisham:p1–p2, h_7478_983_ra_umar_hisham:p1 | عمر، من طريق نافع عن ابن عمر]]

وبلغ الرجلان المدينة فنزلا بقباء. وهناك جاء أبو جهل وأخوه الحارث في طلب عيّاش، وكان ابنَ عمّهما وأخاهما لأمّهما، فقالا له: «إنّ أمّك قد نذرت أن لا يمسّ رأسها مُشطٌ حتى تراك، ولا تستظلّ من شمس حتى تراك». فحذّره عمر: «يا عيّاش، إنه والله إن يريدك القوم إلا ليفتنوك عن دينك فاحذرهم، فوالله لو قد آذى أمَّك القملُ لامتشطت، ولو قد اشتدّ عليها حرُّ مكة لاستظلّت». فقال عيّاش: «أَبَرُّ قسمَ أمّي، ولي هنالك مالٌ فآخذه». فلمّا لم يُمسكه حتى نصفُ مال عمر، أعطاه عمر ناقته وقال: «فإن رابك من القوم ريبٌ فانجُ عليها».[[h_23833_497_prep_786:p1, h_23833_498_prep_cont:p1 | عمر]]

وفي بعض الطريق سأله أبو جهل أن يُعقِبه على تلك الناقة، فقد استغلظ بعيره. فما إن أناخوا رواحلهم واستووا بالأرض حتى عدا الأخوان على عيّاش، فأوثقاه، ودخلا به مكة.[[h_23833_497_prep_786:p1, h_23833_498_prep_cont:p1 | عمر]] وحدّث بعضُ آل عيّاش أنهما دخلا به مكة نهارًا موثقًا، ثم قالا: «يا أهل مكة، هكذا فافعلوا بسفهائكم، كما فعلنا بسفيهنا هذا».[[h_23833_498_prep_cont:p2–p3 | ابن إسحاق عن بعض آل عيّاش]]

وكان المهاجرون يقولون إنّ الله لا يقبل ممّن افتُتن توبة، وكان المفتونون يقولون ذلك لأنفسهم. ثم نزلت بالمدينة آيات: ﴿قُلْ يَا عِبَادِيَ الَّذِينَ أَسْرَفُوا عَلَى أَنفُسِهِمْ لَا تَقْنَطُوا مِن رَّحْمَةِ اللَّهِ﴾. فكتبها عمر بيده في صحيفة وبعث بها إلى هشام، فجعل هشام يقرؤها بذي طُوى، يُصعِّد بها في الوادي ويُصوِّب، حتى فهم أنها إنما نزلت فيهم وفي أمثالهم. فرجع إلى بعيره فجلس عليه، ولحق برسول الله ﷺ.[[h_23833_498_prep_787:p1–p3, h_23833_499_prep_cont:p1–p4, h_7478_983_ra_umar_hisham:p1–p4 | عمر، ثم هشام بن العاص]]

وأمّا صهيب فدفع ثمنًا من نوع آخر. ففيما بلغ أبا عثمان النهدي، أبت قريش أن تدعه يخرج بالمال الذي كسبه بينهم، فجعل لهم ماله كلّه على أن يُخلّوا سبيله. فلمّا بلغ ذلك رسول الله ﷺ قال: «ربح صهيب، ربح صهيب».[[h_23833_500_prep_791:p2–p3 | ابن هشام: «ذُكر لي»، عن أبي عثمان النهدي: «بلغني»]]

## دارٌ بين الأنصار

وفي مكة رقّ لأمّ سلمة آخرَ الأمر رجلٌ من قومها، فكلّم بني المغيرة في إطلاقها. فرُدّ إليها ابنها، فارتحلت بعيرها وحدها وابنها في حجرها. فلمّا كانت بالتنعيم لقيها عثمان بن طلحة، فقال: «إلى أين يا بنت أبي أمية؟» قالت: «أريد زوجي بالمدينة». قال: «أوَما معك أحد؟» قالت: «لا والله، إلا اللهُ وبُنَيَّ هذا». قال: «والله ما لكِ من مَترَك»، وأخذ بخطام البعير.[[h_23833_492_prep_cont:p1–p2 | أم سلمة، من طريق ابن إسحاق عن أبيه عن سلمة بن عبد الله]] فما زال يقودها من منزل إلى منزل، يستأخر عنها كلّما ركبت أو نزلت، حتى بلغ بها قرية بني عمرو بن عوف بقباء، فقال: «زوجكِ في هذه القرية، فادخليها على بركة الله»، ثم انصرف راجعًا إلى مكة.[[h_23833_492_prep_cont:p1–p2 | أم سلمة، من طريق ابن إسحاق عن أبيه عن سلمة بن عبد الله]][[h_23833_493_prep_cont:p1 | أم سلمة]] وكانت تقول: «والله ما أعلم أهلَ بيتٍ في الإسلام أصابهم ما أصاب آلَ أبي سلمة، وما رأيت صاحبًا قطّ كان أكرم من عثمان بن طلحة».[[h_23833_493_prep_cont:p2 | أم سلمة، من طريق ابن إسحاق]] ولم يكن يومئذٍ قد أسلم، وإنما أسلم في هدنة الحديبية.[[h_23833_493_prep_foot_3:p1]]

وتتابع المهاجرون، فنزلوا على الأنصار، أهلِ المدينة، في دورهم، فآوَوهم وآسَوهم؛[[h_146_238_early_migrants:p1 | الواقدي]] بل اقترع الأنصار على سُكنى المهاجرين.[[h_1681_5922_hs_billeting_1:p2 | أم العلاء]] وكان سالم مولى أبي حذيفة يؤمّ المهاجرين بقباء قبل أن يقدم رسول الله ﷺ، لأنه كان أكثرهم قرآنًا.[[h_146_238_early_migrants:p1, h_7478_983_ra_salim:p1, h_7478_984_ra_salim:p1–p2, h_930_747_prep_salim:p1–p2 | سالم بقباء؛ ابن عمر؛ ابن عمر من طريق يعقوب بن سفيان]]

وفي مكة كلِبت قريش واغتاظت على من خرج من فتيانها.[[h_146_238_early_migrants:p1 | الواقدي]] ولم يبقَ بها آخرَ الأمر من المؤمنين إلا رسول الله ﷺ وأبو بكر وعليّ، ومَن حُبس وفُتن، والمريض، والضعيف عن الخروج.[[h_146_238_early_migrants:p1 | الواقدي]] وأقام رسول الله ﷺ بمكة ينتظر أن يأذن له ربّه باللحاق بهم.[[h_9783_994_early_migrants:p1]]

## In depth

### مَن كان الأول؟

تجيب الروايات هنا عن أسئلة مختلفة. فابن إسحاق يجعل أبا سلمة أوّلَ من هاجر إلى المدينة من أصحاب رسول الله ﷺ من قريش، ويؤرّخ هجرته قبل بيعة العقبة بسنة؛ ويجعله ابن سعد كذلك أوّلَ من قدمها.[[h_23833_491_prep_780:p1, h_930_737_prep_abu_salama_date:p1 | ابن إسحاق]][[h_146_237_permission:p1–p2 | الواقدي، عن الزهري وعن عروة عن عائشة]] أمّا البراء بن عازب، وهو من أهل المدينة، فيقول: «أوّل من قدم علينا مصعب بن عمير وابن أمّ مكتوم»، وهو يتحدّث عمّن رأى قومُه قدومَه، وعن رجلين كانا يُقرئان الناس القرآن هناك.[[h_930_743_prep_baraa_2:p1, h_930_744_prep_baraa_2:p1, h_1681_5917_hs_early_short_1:p1–p2 | البراء بن عازب، من طريق البخاري]] وأمّا أوّل ظعينة، أي أوّل امرأة قدمت المدينة في هودجها، فيسمّي ابن سعد ليلى بنت أبي حثمة، امرأة عامر بن ربيعة، وقد قدمت في إثر أبي سلمة. ويقول موسى بن عقبة عن الزهري: «ويقال: أوّل ظعينة قدمت المدينة أمّ سلمة. ويقول بعض الناس أمّ عبد الله، والله أعلم»، يعني أمّ عبد الله بنت أبي حثمة.[[h_146_237_permission:p1–p2 | الواقدي، عن الزهري وعن عروة عن عائشة]][[h_7478_980_ra_musa_early:p1–p2, h_7478_981_ra_musa_early:p1 | موسى بن عقبة عن الزهري]][[h_23833_493_prep_781:p1 | ابن إسحاق]] وحديث أمّ سلمة نفسها يدلّ على أنها لم تخرج مع زوجها حين خرج أوّل مرة.[[h_23833_492_prep_cont:p1–p2 | أم سلمة، من طريق ابن إسحاق عن أبيه عن سلمة بن عبد الله]]

### متى خرجوا؟

في رواية ابن إسحاق خرج أبو سلمة مبكرًا، حين آذته قريش بعد رجوعه من الحبشة، قبل البيعة بسنة. أمّا ابن سعد وموسى بن عقبة فيحدّثان عن الخروج العامّ: لمّا اشتدّت قريش على المؤمنين أمرهم النبي ﷺ بالخروج، فخرجوا أرسالًا.[[h_23833_491_prep_780:p1, h_930_737_prep_abu_salama_date:p1 | ابن إسحاق]][[h_146_237_permission:p1–p2 | الواقدي، عن الزهري وعن عروة عن عائشة]][[h_7478_980_ra_musa_early:p1–p2, h_7478_981_ra_musa_early:p1 | موسى بن عقبة عن الزهري]] ورواية ابن سعد يأخذها عن شيخه الواقدي، الذي جمع فيها ما رواه الزهري وعروة عن عائشة. ويضيف موسى بن عقبة أنّ ناسًا من الأصحاب، منهم سعد بن أبي وقاص، مكثوا بمكة حتى قدموا المدينة بعد مقدم النبي ﷺ؛ وينبّه ابن كثير إلى أنّ رواية البراء في صحيح مسلم تصرّح بأنّ سعدًا هاجر قبله، ويقول: «والصواب ما تقدّم».[[h_7478_982_ra_musa_early:p1 | موسى بن عقبة عن الزهري]][[h_930_744_prep_saad_date:p1]]

### ما كان من أمر عيّاش وهشام؟

الآيات التي بعث بها عمر هي من سورة الزمر (٥٣–٥٥)؛ وفي رواية عمر يقرؤها هشام فيفهمها ويلحق بالمدينة.[[h_23833_498_prep_787:p1–p3, h_23833_499_prep_cont:p1–p4, h_7478_983_ra_umar_hisham:p1–p4 | عمر، ثم هشام بن العاص]] ويضيف ابن هشام، عمّن يثق به، أنّ الوليد بن الوليد خرج بعد ذلك إلى مكة مستخفيًا، فقطع قيدَي الرجلين وقدم بهما المدينة.[[h_23833_499_prep_788:p1–p4 | ابن هشام: «حدّثني من أثق به»]] وفي رواية موسى بن عقبة بقي عيّاش موثقًا حتى خرج مع من خرج قبل فتح مكة، وكان رسول الله ﷺ يدعو له بالخلاص.[[h_7478_980_ra_musa_early:p1–p2, h_7478_981_ra_musa_early:p1 | موسى بن عقبة عن الزهري]]

### روايتان في خبر صهيب

الخبر المرويّ أعلاه عن أبي عثمان النهدي، وقد بلغه بواسطة: فابن هشام لا يقول إلا «ذُكر لي»، وأبو عثمان يقول «بلغني».[[h_23833_500_prep_791:p2–p3 | ابن هشام: «ذُكر لي»، عن أبي عثمان النهدي: «بلغني»]] أمّا رواية صهيب نفسه، من طريق سعيد بن المسيّب، التي أخرجها البيهقي ونقلها ابن كثير، فتجعل القصة في وقت لاحق: كان قد همّ بالخروج مع النبي ﷺ وأبي بكر، فصدّه فتيان من قريش؛ فتسلّل منهم ثم أُدرك، فافتدى نفسه بأواقٍ من ذهب مدفونة تحت أُسكُفّة بابه، ثم لحق بالنبي ﷺ بقباء. وهناك قال له النبي ﷺ: «يا أبا يحيى، ربح البيع».[[h_930_745_prep_suhayb:p1–p2, h_930_746_prep_suhayb:p1 | صهيب، من طريق البيهقي]]

### أين نزلوا، ومن أمّهم في الصلاة

يذكر ابن إسحاق منازل المهاجرين. فقد نزل أبو سلمة وعامر بن ربيعة وابنا جحش على مبشّر بن عبد المنذر بقباء، في بني عمرو بن عوف.[[h_23833_495_prep_cont:p1 | ابن إسحاق]] ونزل عبد الرحمن بن عوف على سعد بن الربيع، ومصعب بن عمير على سعد بن معاذ، وعثمان بن عفان على أوس بن ثابت أخي حسان بن ثابت؛ قال ابن إسحاق: فلذلك كان حسان يحبّ عثمان ويبكيه حين قُتل.[[h_23833_502_prep_794:p1, h_23833_502_prep_796:p1, h_23833_502_prep_798:p1 | ابن إسحاق]]

ويذكر ابن سعد أنّ سالمًا كان يؤمّ المهاجرين بقباء. أمّا ابن عمر، في روايتين أخرجهما البيهقي ويعقوب بن سفيان، فيتحدّث عن جماعة أصغر: نزل عمر وأبو عبيدة وسالم موضعًا يسمّيه العُصبة، فكان يؤمّهم سالم لأنه كان أكثرهم قرآنًا.[[h_146_238_early_migrants:p1, h_7478_983_ra_salim:p1, h_7478_984_ra_salim:p1–p2, h_930_747_prep_salim:p1–p2 | سالم بقباء؛ ابن عمر؛ ابن عمر من طريق يعقوب بن سفيان]]

### «ابن أخي هذا»

قال أبو جهل كلمته عند الدار الخالية والعباسُ إلى جانبه، والنبي ﷺ ابنُ أخي العباس. ويورد ابن إسحاق الكلام دون أن يسمّي من عنى بقوله «هذا».[[h_23833_493_prep_781:p1, h_23833_494_prep_cont:p1–p7 | ابن إسحاق: بنو جحش، وأبو أحمد، والدار الخالية، وعتبة وأبو جهل]]
